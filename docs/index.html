ESP32-S3 FreeRTOS Multi-Core Firmware

Industrial Doctor v3.0

Real-time acoustic machine health monitoring on the edge.

A high-performance embedded systems project built around an ESP32-S3 microcontroller and an INMP441 I2S digital microphone. Utilizes FreeRTOS task partitioning across dual cores, online statistical feature extraction, Welford's variance-based Z-score anomaly engine, and deterministic physics rule validation to diagnose motor health live on a 128×64 OLED display.

⚡ FreeRTOS Dual-Core 🎙️ INMP441 I2S DMA 📊 8-Feature Extraction 🧠 Hybrid Z-Score Voting 📺 SSD1306 OLED UI

Source Code Modules (`src/` & `include/`)

Core Firmware Architecture Breakdown

Detailed technical breakdown of the 5 core C++ modules orchestrating the embedded system.

⚡ main.cpp

Orchestrator & RTOS Controller: Initializes Serial, I2C OLED (0x3C/0x3D), plays boot animation, runs two-stage calibration (Ambient & Healthy Motor), creates queues/mutexes, and spawns TaskAcquire (Core 0), TaskProcess (Core 1), and TaskUI (Core 1).

🎙️ feature_extraction.cpp

Signal Processing Engine: Configures I2S driver pins (WS:13, SD:11, SCK:12) at 16kHz. Processes 1,024 sample frames with normalization, Hann windowing, Goertzel frequency scanning (20Hz–7.5kHz), and time-domain statistical moments.

🧠 anomaly_engine.cpp

Statistical Baseline: Implements Welford’s online mean and variance algorithm during 150-sample healthy calibration. Calculates real-time Z-scores across all 8 features and maps average deviation to a 0.0–1.0 anomaly probability.

⚖️ diagnosis_voter.cpp

Hybrid Decision Ensemble: Evaluates strict fault priority rules (Silent/Off → Ambient Noise → Bearing Impact via Kurtosis/Crest → Friction via Centroid/ZCR → Unbalance via RMS Stability), blending 40% physics rules with 60% anomaly score.

🖥️ inspection_menu.cpp

Interactive OLED UI: Handles joystick ADC input (>3000 UP, <1000 DOWN) with 250ms debouncing. Manages screen navigation between the primary diagnosis dashboard and the scrolling 8-feature live telemetry view.

🚀 boot_animation.h

Startup Sequence: Renders vector-based logo graphics, bezier swooshes, and EKG heartbeat waveforms on the SSD1306 display during device initialization.

void setup() {
    Serial.begin(115200);
    Wire.begin();
    
    // OLED Detection & Boot Animation
    if(!display.begin(SSD1306_SWITCHCAPVCC, 0x3C)) {

    // Initialize I2S Microphone & Calibration Wizard

    runAmbientCalibration();
    runHealthyMotorCalibration();

    // Create FreeRTOS Queues & Mutexes
    xAudioQueue = xQueueCreate(3, sizeof(MotorFeatures));
    xDataMutex = xSemaphoreCreateMutex();

    // Pin Tasks to Cores

    xTaskCreatePinnedToCore(TaskProcess, "ProcessTask", 4096, NULL, 2, &TaskProcessHandle, 1);
    xTaskCreatePinnedToCore(TaskUI, "UITask", 4096, NULL, 1, &TaskUIHandle, 1);
}
cpp src/feature_extraction.cpp - I2S DMA Frame Read & RMS / Kurtosis Calculation 
MotorFeatures FeatureExtractor::getNextFeatures() {
    size_t bytesRead = 0;
    i2s_read(I2S_NUM_0, dmaBuffer, sizeof(dmaBuffer), &bytesRead, portMAX_DELAY);
    int numSamples = bytesRead / sizeof(int32_t);

    float sumSq = 0.0f;
    float m4 = 0.0f;
    float maxAbs = 0.0f;

    for (int i = 0; i < numSamples; i++) {
        float sample = (float)dmaBuffer[i] / 2147483648.0f; // Normalize 32-bit
        sumSq += sample * sample;
        float absVal = fabsf(sample);
        if (absVal > maxAbs) maxAbs = absVal;
    }

    float rms = sqrtf(sumSq / numSamples);
    float variance = sumSq / numSamples;
    float kurtosis = (variance > 0.0f) ? (m4 / numSamples) / (variance * variance) : 0.0f;

    MotorFeatures f;
    f.rms = rms;
    f.crestFactor = (rms > 0.0f) ? (maxAbs / rms) : 0.0f;
    f.kurtosis = kurtosis;
    return f;
}
cpp src/anomaly_engine.cpp - Welford's Online Variance & Z-Score Engine 
void AnomalyEngine::addSample(const float* features) {
    _sampleCount++;
    for (int i = 0; i < 8; i++) {
        float x = features[i];
        float delta = x - _mean[i];
        _mean[i] += delta / _sampleCount;
        float delta2 = x - _mean[i];
        _m2[i] += delta * delta2;
    }
}

float AnomalyEngine::score(const float* features) {
    float totalZ = 0.0f;
    for (int i = 0; i < 8; i++) {
        float stdDev = sqrtf(_m2[i] / (_sampleCount - 1));
        if (stdDev < 1e-3f) stdDev = 1e-3f; // Epsilon guard
        float z = fabsf(features[i] - _mean[i]) / stdDev;
        totalZ += fminf(z, 6.0f); // Cap outlier impact
    }
    float avgZ = totalZ / 8.0f;
    return fminf(fmaxf((avgZ - 2.0f) / 3.0f, 0.0f), 1.0f);
}
cpp src/diagnosis_voter.cpp - Hybrid Physics & Anomaly Ensemble Voting 
DiagnosisResult DiagnosisVoter::voteEnsemble(const float* features, float anomalyScore) {
    float rms = features[0];
    float crest = features[4];
    float kurtosis = features[3];

    DiagnosisResult res;

    // 1. Priority Fault Checks (Physics Rules)
    if (crest > 4.10f || kurtosis > 3.40f) {
        res.status = STATUS_FAULTY;
        res.faultType = FAULT_BEARING;
        res.label = "Bearing Damage";
        res.confidence = 0.92f;
        return res;
    }

    // 2. Hybrid Ensemble Scoring
    float physicsScore = (rms > ambientThreshold) ? 0.1f : 0.8f;
    float finalAnomaly = (physicsScore * 0.40f) + (anomalyScore * 0.60f);

    if (finalAnomaly <= 0.35f) {
        res.status = STATUS_HEALTHY;
        res.label = "Motor Healthy";
        res.confidence = 1.0f - finalAnomaly;
    } else {
        res.status = STATUS_FAULTY;
        res.label = "System Anomaly";
        res.confidence = finalAnomaly;
    }
    return res;
}

        res.status = STATUS_FAULTY;
        res.label = "System Anomaly";
        res.confidence = finalAnomaly;
    }
    return res;
}
